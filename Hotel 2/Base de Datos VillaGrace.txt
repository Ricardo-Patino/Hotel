-- 3) Tablas base
/* ------------------------------------------------------------------------- */
/* Tabla: Habitacion */
CREATE TABLE Habitacion (
    Codigo_Habitacion   INT NOT NULL AUTO_INCREMENT,
    Numero_Habitacion   VARCHAR(10) NOT NULL,
    Tipo                ENUM('Sencilla','Doble','Suite') NOT NULL,
    Precio_Noche        DECIMAL(12,2) NOT NULL,
    Estado              ENUM('Disponible','Ocupada','Mantenimiento') NOT NULL DEFAULT 'Disponible',
    Fecha_modificacion  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT PK_Habitacion PRIMARY KEY (Codigo_Habitacion),
    CONSTRAINT UQ_Habitacion_Numero UNIQUE (Numero_Habitacion),
    KEY IX_Habitacion_Tipo (Tipo),
    KEY IX_Habitacion_Estado (Estado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* Tabla: Funcionario */
CREATE TABLE Funcionario (
    Codigo_Funcionario  INT NOT NULL AUTO_INCREMENT,
    Cedula              VARCHAR(20) NOT NULL,
    Nombre              VARCHAR(50) NOT NULL,
    Apellido            VARCHAR(50) NOT NULL,
    Puesto              VARCHAR(50) NOT NULL,
    Fecha_Nacimiento    DATE NOT NULL,
    Fecha_modificacion  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT PK_Funcionario PRIMARY KEY (Codigo_Funcionario),
    KEY IX_Funcionario_Cedula (Cedula)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* Tabla: Cliente */
CREATE TABLE Cliente (
    Codigo_Cliente      INT NOT NULL AUTO_INCREMENT,
    Cedula              VARCHAR(20) NOT NULL,
    Nombre              VARCHAR(50) NOT NULL,
    Apellido            VARCHAR(50) NOT NULL,
    Telefono            VARCHAR(20) NOT NULL,
    Correo              VARCHAR(120),
    Fecha_Nacimiento    DATE NOT NULL,
    Fecha_modificacion  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT PK_Cliente PRIMARY KEY (Codigo_Cliente),
    KEY IX_Cliente_Correo (Correo),
    KEY IX_Cliente_Cedula (Cedula)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* Tabla: Rol */
CREATE TABLE Rol (
    Codigo_Rol          INT NOT NULL AUTO_INCREMENT,
    Nombre              VARCHAR(30) NOT NULL UNIQUE,
    Descripcion         VARCHAR(200),
    Estado              ENUM('Activo','Inactivo') DEFAULT 'Activo',
    Fecha_Creacion      DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    Fecha_Modificacion  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT PK_Rol PRIMARY KEY (Codigo_Rol)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* Tabla: Usuario  (con FK a Rol y (opcional) a Cliente) */
CREATE TABLE Usuario (
    Codigo_Usuario     INT NOT NULL AUTO_INCREMENT,
    Nombre             VARCHAR(100) NOT NULL,
    Cedula_Pasaporte   VARCHAR(40)  UNIQUE,
    Correo             VARCHAR(120) NOT NULL UNIQUE,
    Telefono           VARCHAR(25)  NOT NULL,
    Contrasena         VARCHAR(255) NOT NULL,
    Rol_Id             INT NULL,
    Estado             ENUM('Activo','Inactivo') DEFAULT 'Activo',
    Fecha_Creacion     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    Fecha_Modificacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    Codigo_Cliente     INT NULL,
    CONSTRAINT PK_Usuario PRIMARY KEY (Codigo_Usuario),
    CONSTRAINT FK_Usuario_Rol
        FOREIGN KEY (Rol_Id) REFERENCES Rol(Codigo_Rol)
        ON UPDATE CASCADE ON DELETE SET NULL,
    CONSTRAINT FK_Usuario_Cliente
        FOREIGN KEY (Codigo_Cliente) REFERENCES Cliente(Codigo_Cliente)
        ON UPDATE CASCADE ON DELETE SET NULL,
    KEY IX_Usuario_Rol (Rol_Id),
    KEY IX_Usuario_Cli (Codigo_Cliente)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* Tabla: Reserva (incluye Huespedes y Observaciones para que coincida con la app) */
CREATE TABLE Reserva (
    Codigo_Reserva       INT NOT NULL AUTO_INCREMENT,
    Codigo_Cliente       INT NOT NULL,
    Codigo_Habitacion    INT NOT NULL,
    Codigo_Funcionario   INT NOT NULL, -- quién registró la reserva
    Fecha_Entrada        DATE NOT NULL,
    Fecha_Salida         DATE NOT NULL,
    Monto_Total          DECIMAL(12,2) NOT NULL,
    Estado               ENUM('Confirmada','Cancelada','Pendiente') NOT NULL DEFAULT 'Confirmada',
    Canal                VARCHAR(30) NULL,
    Numero_Comprobante   VARCHAR(40) NULL,
    Fuente               VARCHAR(30) NULL,
    Politica_Cancelacion VARCHAR(200) NULL,
    Observaciones        VARCHAR(255) NULL,
    Huespedes            VARCHAR(50)  NULL,   -- usado por la app para mostrar/editar
    Descuento_Id         VARCHAR(50)  NULL,
    Fecha_Registro       DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT PK_Reserva PRIMARY KEY (Codigo_Reserva),
    CONSTRAINT FK_Reserva_Cliente
        FOREIGN KEY (Codigo_Cliente) REFERENCES Cliente(Codigo_Cliente)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT FK_Reserva_Habitacion
        FOREIGN KEY (Codigo_Habitacion) REFERENCES Habitacion(Codigo_Habitacion)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT FK_Reserva_Funcionario
        FOREIGN KEY (Codigo_Funcionario) REFERENCES Funcionario(Codigo_Funcionario)
        ON UPDATE CASCADE ON DELETE RESTRICT,
    CONSTRAINT UQ_Reserva_Comprobante UNIQUE (Numero_Comprobante),
    KEY IX_Reserva_Cliente (Codigo_Cliente),
    KEY IX_Reserva_Fechas (Fecha_Entrada, Fecha_Salida),
    KEY IX_Reserva_Estado (Estado)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* Tablas opcionales para documentación de reserva */
CREATE TABLE Documento (
    Id              INT NOT NULL AUTO_INCREMENT,
    Tipo            VARCHAR(50) NOT NULL,
    Ruta            VARCHAR(255) NOT NULL,
    MimeType        VARCHAR(100) NOT NULL DEFAULT 'application/pdf',
    TamanoBytes     INT NULL,
    Fecha_Creacion  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT PK_Documento PRIMARY KEY (Id),
    KEY IX_Documento_Tipo (Tipo)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE ReservaDocumento (
    Id               INT NOT NULL AUTO_INCREMENT,
    Codigo_Reserva   INT NOT NULL,
    Documento_Id     INT NOT NULL,
    CONSTRAINT PK_ReservaDocumento PRIMARY KEY (Id),
    CONSTRAINT FK_ResDoc_Reserva
        FOREIGN KEY (Codigo_Reserva) REFERENCES Reserva(Codigo_Reserva)
        ON UPDATE CASCADE ON DELETE CASCADE,
    CONSTRAINT FK_ResDoc_Documento
        FOREIGN KEY (Documento_Id) REFERENCES Documento(Id)
        ON UPDATE CASCADE ON DELETE CASCADE,
    KEY IX_ResDoc_Reserva (Codigo_Reserva),
    KEY IX_ResDoc_Documento (Documento_Id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* Tablas para KPIs y Auditoría (GRR-01-002) */
CREATE TABLE KPI_Stats (
  Id             INT NOT NULL AUTO_INCREMENT,
  Periodo        ENUM('day','week','month') NOT NULL,
  Clave          VARCHAR(16) NOT NULL,  -- 'YYYY-MM-DD' | 'YYYY-Www' | 'YYYY-MM'
  Total_Reservas INT NOT NULL DEFAULT 0,
  Total_Monto    DECIMAL(14,2) NOT NULL DEFAULT 0.00,
  Fecha_Ultima   DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT PK_KPI_Stats PRIMARY KEY (Id),
  CONSTRAINT UQ_KPI UNIQUE (Periodo, Clave),
  KEY IX_KPI_Periodo (Periodo),
  KEY IX_KPI_Clave (Clave)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE Audit_Log (
  Id        INT NOT NULL AUTO_INCREMENT,
  Fecha     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  Usuario   VARCHAR(120),
  Accion    VARCHAR(60) NOT NULL,
  Detalles  JSON NULL,
  CONSTRAINT PK_Audit_Log PRIMARY KEY (Id),
  KEY IX_Audit_Accion (Accion),
  KEY IX_Audit_Usuario (Usuario),
  KEY IX_Audit_Fecha (Fecha)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

DELIMITER $$
CREATE TRIGGER trg_auditoria_log_after_ins
AFTER INSERT ON Auditoria_Log
FOR EACH ROW
BEGIN
  INSERT INTO Audit_Log (Fecha, Usuario, Accion, Detalles)
  VALUES (
    COALESCE(NEW.Fecha, NOW()),
    NEW.Usuario,
    -- formateo de acción, puedes ajustar si prefieres conservar tal cual
    CONCAT(NEW.Entidad, '.', LOWER(NEW.Accion)),
    -- Detalles como JSON consistente
    JSON_OBJECT(
      'Entidad', NEW.Entidad,
      'Entidad_Id', NEW.Entidad_Id,
      'Datos', NEW.Datos
    )
  );
END$$
DELIMITER ;

-- Vista útil para reportes rápidos
CREATE OR REPLACE VIEW v_reservas_ext AS
SELECT 
    r.Codigo_Reserva, 
    r.Numero_Comprobante,
    r.Estado,
    r.Canal,
    r.Fecha_Entrada,
    r.Fecha_Salida,
    r.Monto_Total,
    r.Fecha_Registro,
    r.Observaciones,
    r.Huespedes,
    c.Codigo_Cliente,
    c.Correo          AS ClienteCorreo,
    CONCAT(c.Nombre,' ',c.Apellido) AS ClienteNombre,
    h.Codigo_Habitacion,
    h.Numero_Habitacion,
    h.Tipo            AS Tipo_Habitacion,
    h.Precio_Noche
FROM Reserva r
JOIN Cliente c     ON r.Codigo_Cliente    = c.Codigo_Cliente
JOIN Habitacion h  ON r.Codigo_Habitacion = h.Codigo_Habitacion;

    r.Estado, r.Canal, r.Numero_Comprobante,

JOIN Habitacion h  ON r.Codigo_Habitacion = h.Codigo_Habitacion
ORDER BY r.Fecha_Entrada DESC;

/* 1.2) Cabecera de transacción de libro mayor (una por transacción POS) */
CREATE TABLE IF NOT EXISTS fin_ledger_tx (
  id_tx        INT NOT NULL AUTO_INCREMENT,
  external_id  VARCHAR(64) NOT NULL,                -- ID de la transacción en el POS (único)
  source       ENUM('POS','BACKOFFICE') NOT NULL DEFAULT 'POS',
  reserva_id   INT NULL,
  currency     VARCHAR(10) NOT NULL DEFAULT 'CRC',
  total        DECIMAL(14,2) NOT NULL DEFAULT 0.00, -- total de la transacción
  status       ENUM('posted','voided') NOT NULL DEFAULT 'posted',
  created_at   DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  meta         JSON NULL,
  CONSTRAINT PK_fin_ledger_tx PRIMARY KEY (id_tx),
  CONSTRAINT UQ_fin_ledger_tx_ext UNIQUE (external_id),
  KEY IX_fin_ledger_tx_reserva (reserva_id),
  KEY IX_fin_ledger_tx_status (status)
  -- Si quieres FK dura, descomenta:
  -- ,CONSTRAINT FK_fin_ledger_tx_reserva
  --   FOREIGN KEY (reserva_id) REFERENCES Reserva(Codigo_Reserva)
  --   ON UPDATE CASCADE ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* 1.3) Líneas contables (doble partida) para cada transacción */
CREATE TABLE IF NOT EXISTS fin_ledger_line (
  id_line    INT NOT NULL AUTO_INCREMENT,
  id_tx      INT NOT NULL,
  line_no    INT NOT NULL,                           -- 1,2,3...
  account    VARCHAR(64) NOT NULL,                   -- p.ej. 'VENTA_HAB', 'IVA', 'TPV', 'CAJA'
  debit      DECIMAL(14,2) NOT NULL DEFAULT 0.00,
  credit     DECIMAL(14,2) NOT NULL DEFAULT 0.00,
  description VARCHAR(255) NULL,
  CONSTRAINT PK_fin_ledger_line PRIMARY KEY (id_line),
  CONSTRAINT UQ_fin_ledger_line UNIQUE (id_tx, line_no),
  KEY IX_fin_ledger_line_account (account),
  CONSTRAINT FK_fin_ledger_line_tx
    FOREIGN KEY (id_tx) REFERENCES fin_ledger_tx(id_tx)
    ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* (Opcional) Vista rápida por reserva: total pagado vía POS */
CREATE OR REPLACE VIEW v_reserva_pagos_pos AS
SELECT
  t.reserva_id,
  SUM(CASE WHEN l.debit > 0 THEN l.debit ELSE 0 END) AS total_pagado_pos
FROM fin_ledger_tx t
JOIN fin_ledger_line l ON l.id_tx = t.id_tx
WHERE t.status = 'posted'
GROUP BY t.reserva_id;

-- 3) Vista de conciliación diaria (esperado vs contado y descuadre)
CREATE OR REPLACE VIEW v_fin_caja_resumen AS
SELECT
  s.id_session,
  s.fecha,
  s.opening_cash,
  IFNULL(r.ingresos_efectivo, 0.00) AS ingresos_efectivo,
  IFNULL(m.egresos_efectivo, 0.00)  AS egresos_efectivo,
  IFNULL(m.ajustes_mas, 0.00)       AS ajustes_mas,
  IFNULL(m.ajustes_menos, 0.00)     AS ajustes_menos,
  (s.opening_cash + IFNULL(r.ingresos_efectivo,0) - IFNULL(m.egresos_efectivo,0)
    + IFNULL(m.ajustes_mas,0) - IFNULL(m.ajustes_menos,0)) AS efectivo_esperado,
  s.closing_cash_counted AS efectivo_contado,
  (IFNULL(s.closing_cash_counted,0) - 
   (s.opening_cash + IFNULL(r.ingresos_efectivo,0) - IFNULL(m.egresos_efectivo,0)
     + IFNULL(m.ajustes_mas,0) - IFNULL(m.ajustes_menos,0))) AS descuadre
FROM fin_cash_session s
LEFT JOIN (
  SELECT DATE(creado_en) AS fecha, SUM(monto) AS ingresos_efectivo
    FROM fin_receipts
   WHERE estado='Emitido' AND metodo='Efectivo'
   GROUP BY DATE(creado_en)
) r ON r.fecha = s.fecha
LEFT JOIN (
  SELECT DATE(created_at) AS fecha,
         SUM(CASE WHEN tipo='Egreso' AND metodo='Efectivo' AND estado='Aplicado' THEN monto ELSE 0 END) AS egresos_efectivo,
         SUM(CASE WHEN metodo='Efectivo' AND estado='Aplicado' AND monto >= 0 AND tipo IN ('Ingreso','Ajuste') THEN monto ELSE 0 END) AS ajustes_mas,
         SUM(CASE WHEN metodo='Efectivo' AND estado='Aplicado' AND monto < 0  AND tipo='Ajuste' THEN -monto ELSE 0 END) AS ajustes_menos
  FROM fin_cash_movement
  GROUP BY DATE(created_at)
) m ON m.fecha = s.fecha;
