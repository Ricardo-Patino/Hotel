-- Triggers para mantener Habitacion_Id sincronizado
DELIMITER $$
CREATE TRIGGER trg_reserva_sync_before_ins
BEFORE INSERT ON Reserva
FOR EACH ROW
BEGIN
  SET NEW.Habitacion_Id = NEW.Codigo_Habitacion;
END$$

CREATE TRIGGER trg_reserva_sync_before_upd
BEFORE UPDATE ON Reserva
FOR EACH ROW
BEGIN
  IF NEW.Codigo_Habitacion <> OLD.Codigo_Habitacion THEN
    SET NEW.Habitacion_Id = NEW.Codigo_Habitacion;
  END IF;
END$$
DELIMITER ;

DELIMITER $$
CREATE TRIGGER trg_auditoria_log_after_ins
AFTER INSERT ON Auditoria_Log
FOR EACH ROW
BEGIN
  INSERT INTO Audit_Log (Fecha, Usuario, Accion, Detalles)
  VALUES (
    COALESCE(NEW.Fecha, NOW()),
    NEW.Usuario,
    CONCAT(NEW.Entidad, '.', LOWER(NEW.Accion)),
    JSON_OBJECT('Entidad', NEW.Entidad, 'Entidad_Id', NEW.Entidad_Id, 'Datos', NEW.Datos)

CREATE OR REPLACE VIEW v_reservas_ext AS
SELECT 
  r.Codigo_Reserva, 
  r.Numero_Comprobante,
  r.Estado,
  r.Canal,
  r.Fecha_Entrada,
  r.Fecha_Salida,
  r.Monto_Total,
  r.Fecha_Registro,
  r.Observaciones,
  r.Huespedes,
  c.Codigo_Cliente,
  c.Correo          AS ClienteCorreo,
  CONCAT(c.Nombre,' ',c.Apellido) AS ClienteNombre,
  h.Codigo_Habitacion,
  h.Numero_Habitacion,
  h.Tipo            AS Tipo_Habitacion,
  h.Precio_Noche
FROM Reserva r
JOIN Cliente c     ON r.Codigo_Cliente    = c.Codigo_Cliente
JOIN Habitacion h  ON r.Codigo_Habitacion = h.Codigo_Habitacion;

CREATE OR REPLACE VIEW v_reserva_pagos_pos AS
SELECT
  t.reserva_id,
  SUM(CASE WHEN l.debit > 0 THEN l.debit ELSE 0 END) AS total_pagado_pos
FROM fin_ledger_tx t
JOIN fin_ledger_line l ON l.id_tx = t.id_tx
WHERE t.status = 'posted'
GROUP BY t.reserva_id;

CREATE OR REPLACE VIEW v_fin_caja_resumen AS
SELECT
  s.id_session,
  s.fecha,
  s.opening_cash,
  IFNULL(r.ingresos_efectivo, 0.00) AS ingresos_efectivo,
  IFNULL(m.egresos_efectivo, 0.00)  AS egresos_efectivo,
  IFNULL(m.ajustes_mas, 0.00)       AS ajustes_mas,
  IFNULL(m.ajustes_menos, 0.00)     AS ajustes_menos,
  (s.opening_cash + IFNULL(r.ingresos_efectivo,0) - IFNULL(m.egresos_efectivo,0)
    + IFNULL(m.ajustes_mas,0) - IFNULL(m.ajustes_menos,0)) AS efectivo_esperado,
  s.closing_cash_counted AS efectivo_contado,
  (IFNULL(s.closing_cash_counted,0) - 
   (s.opening_cash + IFNULL(r.ingresos_efectivo,0) - IFNULL(m.egresos_efectivo,0)
     + IFNULL(m.ajustes_mas,0) - IFNULL(m.ajustes_menos,0))) AS descuadre
FROM fin_cash_session s
LEFT JOIN (
  SELECT DATE(creado_en) AS fecha, SUM(monto) AS ingresos_efectivo
  FROM fin_receipts
  WHERE estado='Emitido' AND metodo='Efectivo'
  GROUP BY DATE(creado_en)
) r ON r.fecha = s.fecha
LEFT JOIN (
  SELECT DATE(created_at) AS fecha,
         SUM(CASE WHEN tipo='Egreso' AND metodo='Efectivo' AND estado='Aplicado' THEN monto ELSE 0 END) AS egresos_efectivo,
         SUM(CASE WHEN metodo='Efectivo' AND estado='Aplicado' AND monto >= 0 AND tipo IN ('Ingreso','Ajuste') THEN monto ELSE 0 END) AS ajustes_mas,
         SUM(CASE WHEN metodo='Efectivo' AND estado='Aplicado' AND monto < 0  AND tipo='Ajuste' THEN -monto ELSE 0 END) AS ajustes_menos
  FROM fin_cash_movement
  GROUP BY DATE(created_at)
) m ON m.fecha = s.fecha;

   -------------------------- */
CREATE TABLE EvtEvento (
  Id                  INT NOT NULL AUTO_INCREMENT,
  Titulo              VARCHAR(200) NOT NULL,
  Estado              ENUM('BORRADOR','CONFIRMADO','CANCELADO') NOT NULL DEFAULT 'BORRADOR',
  Inicio              DATETIME NOT NULL,
  Fin                 DATETIME NOT NULL,
  TZ                  VARCHAR(64) NOT NULL DEFAULT 'America/Costa_Rica',
  Organizador         VARCHAR(120) NULL,
  Notas               TEXT NULL,
  Salon_Id            INT NULL,
  ReservaEstancia_Id  INT NULL,   -- vínculo con estancia del huésped
  Plantilla_Id        INT NULL,
  Fecha_Creacion      DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  Fecha_Actualizacion DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  CONSTRAINT PK_EvtEvento PRIMARY KEY (Id),
  KEY IX_EvtEvento_Rango (Inicio, Fin),
  KEY IX_EvtEvento_Estado (Estado),
  KEY IX_EvtEvento_Salon (Salon_Id),
  KEY IX_EvtEvento_Estancia (ReservaEstancia_Id),
  CONSTRAINT FK_EvtEvento_Salon
    FOREIGN KEY (Salon_Id) REFERENCES EvtSalon(Id)
    ON UPDATE CASCADE ON DELETE SET NULL,
  CONSTRAINT FK_EvtEvento_Plantilla
    FOREIGN KEY (Plantilla_Id) REFERENCES EvtPlantilla(Id)
    ON UPDATE CASCADE ON DELETE SET NULL,
  CONSTRAINT FK_EvtEvento_Estancia
    FOREIGN KEY (ReservaEstancia_Id) REFERENCES ReservaEstancia(Id_Estancia)
    ON UPDATE CASCADE ON DELETE SET NULL
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* --------------------------
   Recursos por evento (cantidades)
   -------------------------- */
CREATE TABLE EvtEventoRecurso (
  Id         INT NOT NULL AUTO_INCREMENT,
  Evento_Id  INT NOT NULL,
  Recurso_Id INT NOT NULL,
  Cantidad   INT NOT NULL DEFAULT 1,
  CONSTRAINT PK_EvtEventoRecurso PRIMARY KEY (Id),
  CONSTRAINT UQ_EvtEventoRecurso UNIQUE (Evento_Id, Recurso_Id),
  KEY IX_EvtEvRec_Evento (Evento_Id),
  KEY IX_EvtEvRec_Recurso (Recurso_Id),
  CONSTRAINT FK_EvtEvRec_Evento
    FOREIGN KEY (Evento_Id) REFERENCES EvtEvento(Id)
    ON UPDATE CASCADE ON DELETE CASCADE,
  CONSTRAINT FK_EvtEvRec_Recurso
    FOREIGN KEY (Recurso_Id) REFERENCES EvtRecurso(Id)
    ON UPDATE CASCADE ON DELETE RESTRICT
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* --------------------------
   Asistentes y lista de espera del evento
   -------------------------- */
CREATE TABLE EvtAsistente (
  Id          INT NOT NULL AUTO_INCREMENT,
  Evento_Id   INT NOT NULL,
  Nombre      VARCHAR(120) NOT NULL,
  Email       VARCHAR(180) NULL,
  Estado      ENUM('REGISTRADO','CHECKIN','CANCELADO') NOT NULL DEFAULT 'REGISTRADO',
  CheckIn_At  DATETIME NULL,
  CONSTRAINT PK_EvtAsistente PRIMARY KEY (Id),
  CONSTRAINT UQ_EvtAsistente UNIQUE (Evento_Id, Email),
  KEY IX_EvtAsistente_Evento (Evento_Id),
  KEY IX_EvtAsistente_Estado (Estado),
  CONSTRAINT FK_EvtAsistente_Evento
    FOREIGN KEY (Evento_Id) REFERENCES EvtEvento(Id)
    ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE EvtWaitlist (
  Id         INT NOT NULL AUTO_INCREMENT,
  Evento_Id  INT NOT NULL,
  Nombre     VARCHAR(120) NOT NULL,
  Email      VARCHAR(180) NULL,
  Estado     ENUM('EN_ESPERA','INVITADO','DESCARTADO') NOT NULL DEFAULT 'EN_ESPERA',
  CONSTRAINT PK_EvtWaitlist PRIMARY KEY (Id),
  KEY IX_EvtWaitlist_Evento (Evento_Id),
  KEY IX_EvtWaitlist_Estado (Estado),
  CONSTRAINT FK_EvtWaitlist_Evento
    FOREIGN KEY (Evento_Id) REFERENCES EvtEvento(Id)
    ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* --------------------------
   Presupuesto y costos reales
   -------------------------- */
CREATE TABLE EvtPresupuesto (
  Id         INT NOT NULL AUTO_INCREMENT,
  Evento_Id  INT NOT NULL,
  Concepto   VARCHAR(200) NOT NULL,
  Monto      DECIMAL(12,2) NOT NULL DEFAULT 0.00,
  CONSTRAINT PK_EvtPresupuesto PRIMARY KEY (Id),
  CONSTRAINT UQ_EvtPresupuesto UNIQUE (Evento_Id, Concepto),
  KEY IX_EvtPres_Evento (Evento_Id),
  CONSTRAINT FK_EvtPres_Evento
    FOREIGN KEY (Evento_Id) REFERENCES EvtEvento(Id)
    ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

CREATE TABLE EvtCosto (
  Id         INT NOT NULL AUTO_INCREMENT,
  Evento_Id  INT NOT NULL,
  Concepto   VARCHAR(200) NOT NULL,
  Monto      DECIMAL(12,2) NOT NULL DEFAULT 0.00,
  CONSTRAINT PK_EvtCosto PRIMARY KEY (Id),
  KEY IX_EvtCosto_Evento (Evento_Id),
  CONSTRAINT FK_EvtCosto_Evento
    FOREIGN KEY (Evento_Id) REFERENCES EvtEvento(Id)
    ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* --------------------------
   Encuesta post-evento (NPS)
   -------------------------- */
CREATE TABLE EvtEncuesta (
  Id         INT NOT NULL AUTO_INCREMENT,
  Evento_Id  INT NOT NULL,
  NPS        INT NULL,              -- 0..10
  Comentario TEXT NULL,
  Created_At DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT PK_EvtEncuesta PRIMARY KEY (Id),
  KEY IX_EvtEncuesta_Evento (Evento_Id),
  CONSTRAINT FK_EvtEncuesta_Evento
    FOREIGN KEY (Evento_Id) REFERENCES EvtEvento(Id)
    ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* --------------------------
   Notificaciones programadas
   -------------------------- */
CREATE TABLE EvtNotificacion (
  Id            INT NOT NULL AUTO_INCREMENT,
  Evento_Id     INT NOT NULL,
  Tipo          ENUM('7D','24H','2H','CONFIRMACION','UPGRADE') NOT NULL,
  Enviado       TINYINT(1) NOT NULL DEFAULT 0,
  Destinatario  VARCHAR(180) NULL,
  Payload       JSON NULL,
  Programado_At DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  CONSTRAINT PK_EvtNotificacion PRIMARY KEY (Id),
  KEY IX_EvtNotif_Evento (Evento_Id),
  KEY IX_EvtNotif_Enviado (Enviado),
  CONSTRAINT FK_EvtNotif_Evento
    FOREIGN KEY (Evento_Id) REFERENCES EvtEvento(Id)
    ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

/* --------------------------
   Órdenes de trabajo por confirmación
   -------------------------- */
CREATE TABLE EvtWorkOrder (
  Id        INT NOT NULL AUTO_INCREMENT,
  Evento_Id INT NOT NULL,
  Area      ENUM('HOUSEKEEPING','MANTENIMIENTO','COCINA') NOT NULL,
  Tarea     VARCHAR(240) NOT NULL,
  Prioridad ENUM('BAJA','MEDIA','ALTA','URGENTE') NOT NULL DEFAULT 'MEDIA',
  Fecha     DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  Estado    ENUM('PENDIENTE','EN_PROCESO','COMPLETADA','CANCELADA') NOT NULL DEFAULT 'PENDIENTE',
  CONSTRAINT PK_EvtWorkOrder PRIMARY KEY (Id),
  KEY IX_EvtWO_Evento (Evento_Id),
  KEY IX_EvtWO_Area (Area),
  KEY IX_EvtWO_Estado (Estado),
  CONSTRAINT FK_EvtWO_Evento
    FOREIGN KEY (Evento_Id) REFERENCES EvtEvento(Id)
    ON UPDATE CASCADE ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

ALTER TABLE Funcionario
  ADD COLUMN Departamento         VARCHAR(80) NULL AFTER Puesto,
  ADD COLUMN Fecha_Ingreso        DATE NOT NULL DEFAULT (CURDATE()) AFTER Fecha_Nacimiento,
  ADD COLUMN Salario_Base_Mensual DECIMAL(12,2) NOT NULL DEFAULT 0.00 AFTER Fecha_Ingreso,
  ADD COLUMN Estado_Empleado      ENUM('Activo','Inactivo','Suspendido') NOT NULL DEFAULT 'Activo' AFTER Salario_Base_Mensual,
  ADD COLUMN Tipo_Contrato        ENUM('Tiempo completo','Medio tiempo','Servicios profesionales','Temporal') NOT NULL DEFAULT 'Tiempo completo' AFTER Estado_Empleado,
  ADD COLUMN Cuenta_Bancaria      VARCHAR(60) NULL AFTER Tipo_Contrato,
  ADD COLUMN Banco                VARCHAR(60) NULL AFTER Cuenta_Bancaria,
  ADD COLUMN Fecha_mod_alta       DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP AFTER Fecha_modificacion;

-- Índices útiles para búsquedas
ALTER TABLE Funcionario
  ADD INDEX IX_Funcionario_Departamento (Departamento),
  ADD INDEX IX_Funcionario_Estado (Estado_Empleado),
  ADD INDEX IX_Funcionario_FechaIngreso (Fecha_Ingreso);
