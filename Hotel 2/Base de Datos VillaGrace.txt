-- Triggers para mantener Habitacion_Id sincronizado
DELIMITER $$
CREATE TRIGGER trg_reserva_sync_before_ins
BEFORE INSERT ON Reserva
FOR EACH ROW
BEGIN
  SET NEW.Habitacion_Id = NEW.Codigo_Habitacion;
END$$

CREATE TRIGGER trg_reserva_sync_before_upd
BEFORE UPDATE ON Reserva
FOR EACH ROW
BEGIN
  IF NEW.Codigo_Habitacion <> OLD.Codigo_Habitacion THEN
    SET NEW.Habitacion_Id = NEW.Codigo_Habitacion;
  END IF;
END$$
DELIMITER ;

CREATE OR REPLACE VIEW v_reservas_ext AS
SELECT 
  r.Codigo_Reserva, 
  r.Numero_Comprobante,
  r.Estado,
  r.Canal,
  r.Fecha_Entrada,
  r.Fecha_Salida,
  r.Monto_Total,
  r.Fecha_Registro,
  r.Observaciones,
  r.Huespedes,
  c.Codigo_Cliente,
  c.Correo          AS ClienteCorreo,
  CONCAT(c.Nombre,' ',c.Apellido) AS ClienteNombre,
  h.Codigo_Habitacion,
  h.Numero_Habitacion,
  h.Tipo            AS Tipo_Habitacion,
  h.Precio_Noche
FROM Reserva r
JOIN Cliente c     ON r.Codigo_Cliente    = c.Codigo_Cliente
JOIN Habitacion h  ON r.Codigo_Habitacion = h.Codigo_Habitacion;

CREATE OR REPLACE VIEW v_reserva_pagos_pos AS
SELECT
  t.reserva_id,
  SUM(CASE WHEN l.debit > 0 THEN l.debit ELSE 0 END) AS total_pagado_pos
FROM fin_ledger_tx t
JOIN fin_ledger_line l ON l.id_tx = t.id_tx
WHERE t.status = 'posted'
GROUP BY t.reserva_id;

CREATE OR REPLACE VIEW v_fin_caja_resumen AS
SELECT
  s.id_session,
  s.fecha,
  s.opening_cash,
  IFNULL(r.ingresos_efectivo, 0.00) AS ingresos_efectivo,
  IFNULL(m.egresos_efectivo, 0.00)  AS egresos_efectivo,
  IFNULL(m.ajustes_mas, 0.00)       AS ajustes_mas,
  IFNULL(m.ajustes_menos, 0.00)     AS ajustes_menos,
  (s.opening_cash + IFNULL(r.ingresos_efectivo,0) - IFNULL(m.egresos_efectivo,0)
    + IFNULL(m.ajustes_mas,0) - IFNULL(m.ajustes_menos,0)) AS efectivo_esperado,
  s.closing_cash_counted AS efectivo_contado,
  (IFNULL(s.closing_cash_counted,0) - 
   (s.opening_cash + IFNULL(r.ingresos_efectivo,0) - IFNULL(m.egresos_efectivo,0)
     + IFNULL(m.ajustes_mas,0) - IFNULL(m.ajustes_menos,0))) AS descuadre
FROM fin_cash_session s
LEFT JOIN (
  SELECT DATE(creado_en) AS fecha, SUM(monto) AS ingresos_efectivo
  FROM fin_receipts
  WHERE estado='Emitido' AND metodo='Efectivo'
  GROUP BY DATE(creado_en)
) r ON r.fecha = s.fecha
LEFT JOIN (
  SELECT DATE(created_at) AS fecha,
         SUM(CASE WHEN tipo='Egreso' AND metodo='Efectivo' AND estado='Aplicado' THEN monto ELSE 0 END) AS egresos_efectivo,
         SUM(CASE WHEN metodo='Efectivo' AND estado='Aplicado' AND monto >= 0 AND tipo IN ('Ingreso','Ajuste') THEN monto ELSE 0 END) AS ajustes_mas,
         SUM(CASE WHEN metodo='Efectivo' AND estado='Aplicado' AND monto < 0  AND tipo='Ajuste' THEN -monto ELSE 0 END) AS ajustes_menos
  FROM fin_cash_movement
  GROUP BY DATE(created_at)
) m ON m.fecha = s.fecha;

DELIMITER $$
CREATE TRIGGER trg_sac_reserva_after_ins
AFTER INSERT ON Reserva FOR EACH ROW
BEGIN
  DECLARE cli_email VARCHAR(120);
  DECLARE cli_tel   VARCHAR(25);
  DECLARE canal     VARCHAR(10);

  SELECT c.Correo, c.Telefono INTO cli_email, cli_tel
  FROM Cliente c WHERE c.Codigo_Cliente = NEW.Codigo_Cliente LIMIT 1;

  SELECT COALESCE(p.Canal,'email') INTO canal
  FROM SAC_NotifPref p WHERE p.Codigo_Cliente = NEW.Codigo_Cliente LIMIT 1;

  IF canal IN ('email','ambos') AND cli_email IS NOT NULL THEN
    INSERT INTO SAC_Outbox (Canal, Para, Asunto, Cuerpo, Estado, Programado_At, Ref_Entidad, Ref_Id)
    VALUES ('email', cli_email, 'Recordatorio de reserva',
            CONCAT('Su reserva ', COALESCE(NEW.Numero_Comprobante,''), ' se acerca. ',
                   'Check-in desde ', (SELECT Valor FROM SAC_Config WHERE Clave='checkin_inicio' LIMIT 1), 
                   '. Check-out hasta ', (SELECT Valor FROM SAC_Config WHERE Clave='checkout_limite' LIMIT 1), '.'),
            'PENDIENTE', DATE_SUB(NEW.Fecha_Entrada, INTERVAL 48 HOUR),'Reserva', NEW.Codigo_Reserva);
    INSERT INTO SAC_Outbox (Canal, Para, Asunto, Cuerpo, Estado, Programado_At, Ref_Entidad, Ref_Id)
    VALUES ('email', cli_email, 'Recordatorio de reserva',
            CONCAT('Le esperamos mañana. Check-in desde ',
                   (SELECT Valor FROM SAC_Config WHERE Clave='checkin_inicio' LIMIT 1), '.'),
            'PENDIENTE', DATE_SUB(NEW.Fecha_Entrada, INTERVAL 24 HOUR),'Reserva', NEW.Codigo_Reserva);
  END IF;

  IF canal IN ('sms','ambos') AND cli_tel IS NOT NULL THEN
    INSERT INTO SAC_Outbox (Canal, Para, Asunto, Cuerpo, Estado, Programado_At, Ref_Entidad, Ref_Id)
    VALUES ('sms', cli_tel, NULL,
            CONCAT('Villa Grace: su reserva ', COALESCE(NEW.Numero_Comprobante,''), 
                   '. Check-in ', (SELECT Valor FROM SAC_Config WHERE Clave='checkin_inicio' LIMIT 1)),
            'PENDIENTE', DATE_SUB(NEW.Fecha_Entrada, INTERVAL 24 HOUR),'Reserva', NEW.Codigo_Reserva);
  END IF;
END$$
