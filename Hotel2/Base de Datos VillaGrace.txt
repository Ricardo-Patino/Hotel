  Observaciones          VARCHAR(255) NULL,

    ON UPDATE CASCADE ON DELETE RESTRICT

/* ------------------------------------------------------------------------- */
/* Triggers de Reserva (CONSOLIDADOS)                                         */
/*  - BEFORE INSERT: sincroniza Habitacion_Id                                 */
/*  - BEFORE UPDATE: bloquea historial + sincroniza Habitacion_Id             */
/* ------------------------------------------------------------------------- */
DELIMITER $$

CREATE TRIGGER trg_reserva_before_ins
BEFORE INSERT ON Reserva
FOR EACH ROW
BEGIN
  SET NEW.Habitacion_Id = NEW.Codigo_Habitacion;
END$$

CREATE TRIGGER trg_reserva_before_upd
BEFORE UPDATE ON Reserva
FOR EACH ROW
BEGIN
  /* ACE-01-002: bloquear edición si ya es historial */
  IF OLD.Fecha_Salida < CURDATE() THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT='ACE-01-002: Reserva en historial, no editable';
  END IF;

  /* sync */
  IF NEW.Codigo_Habitacion <> OLD.Codigo_Habitacion THEN
    SET NEW.Habitacion_Id = NEW.Codigo_Habitacion;
  END IF;
END$$

DELIMITER ;

CREATE TRIGGER trg_auditoria_log_after_ins
AFTER INSERT ON Auditoria_Log
FOR EACH ROW
BEGIN
  INSERT INTO Audit_Log (Fecha, Usuario, Accion, Detalles)
  VALUES (
    COALESCE(NEW.Fecha, NOW()),
    NEW.Usuario,
    CONCAT(NEW.Entidad, '.', LOWER(NEW.Accion)),
    JSON_OBJECT('Entidad', NEW.Entidad, 'Entidad_Id', NEW.Entidad_Id, 'Datos', NEW.Datos)
  );
END$$

/* Triggers SAC */
DELIMITER $$

/* Al crear reserva: agendar recordatorios por canal preferido (si existe) */
CREATE TRIGGER trg_sac_reserva_after_ins
AFTER INSERT ON Reserva
FOR EACH ROW
BEGIN
  DECLARE cli_email VARCHAR(120);
  DECLARE cli_tel   VARCHAR(25);
  DECLARE canal_pref VARCHAR(10);

  SELECT c.Correo, c.Telefono INTO cli_email, cli_tel
  FROM Cliente c WHERE c.Codigo_Cliente = NEW.Codigo_Cliente LIMIT 1;

  SELECT COALESCE(p.Canal,'email') INTO canal_pref
  FROM SAC_NotifPref p
  WHERE p.Codigo_Cliente = NEW.Codigo_Cliente
  LIMIT 1;

  IF canal_pref IN ('email','ambos') AND cli_email IS NOT NULL THEN
    INSERT INTO SAC_Outbox (Tipo, To_Addr, Subject, Payload_JSON, Estado, Programado_At, Ref_Entidad, Ref_Id, Canal, Para, Asunto, Cuerpo)
    VALUES (
      'EMAIL',
      cli_email,
      'Recordatorio de reserva',
      JSON_OBJECT('reserva', NEW.Codigo_Reserva, 'comprobante', COALESCE(NEW.Numero_Comprobante,'')),
      'PENDIENTE',
      DATE_SUB(CONCAT(NEW.Fecha_Entrada,' 00:00:00'), INTERVAL 48 HOUR),
      'Reserva', NEW.Codigo_Reserva,
      'email', cli_email, 'Recordatorio de reserva',
      CONCAT('Su reserva ', COALESCE(NEW.Numero_Comprobante,''), ' se acerca. ',
             'Check-in desde ', (SELECT Valor FROM SAC_Config WHERE Clave='checkin_inicio' LIMIT 1),
             '. Check-out hasta ', (SELECT Valor FROM SAC_Config WHERE Clave='checkout_limite' LIMIT 1), '.')
    );

    INSERT INTO SAC_Outbox (Tipo, To_Addr, Subject, Payload_JSON, Estado, Programado_At, Ref_Entidad, Ref_Id, Canal, Para, Asunto, Cuerpo)
    VALUES (
      'EMAIL',
      cli_email,
      'Recordatorio de reserva',
      JSON_OBJECT('reserva', NEW.Codigo_Reserva, 'comprobante', COALESCE(NEW.Numero_Comprobante,'')),
      'PENDIENTE',
      DATE_SUB(CONCAT(NEW.Fecha_Entrada,' 00:00:00'), INTERVAL 24 HOUR),
      'Reserva', NEW.Codigo_Reserva,
      'email', cli_email, 'Recordatorio de reserva',
      CONCAT('Le esperamos mañana. Check-in desde ',
             (SELECT Valor FROM SAC_Config WHERE Clave='checkin_inicio' LIMIT 1), '.')
    );

DELIMITER $$
CREATE FUNCTION loyalty_level(p INT)
RETURNS VARCHAR(10)
DETERMINISTIC
BEGIN
  RETURN (CASE
    WHEN p >= 700 THEN 'VIP'
    WHEN p >= 300 THEN 'Gold'
    WHEN p >= 100 THEN 'Silver'
    ELSE 'Ninguno'
  END);
END$$
DELIMITER ;

DELIMITER $$
CREATE TRIGGER trg_reserva_loyalty_after_upd
AFTER UPDATE ON Reserva
FOR EACH ROW
BEGIN
  DECLARE v_noches INT;
  DECLARE v_pts    INT;

  IF NEW.Estado='Confirmada'
     AND NEW.Fecha_Salida <= CURDATE()
     AND NOT EXISTS (SELECT 1 FROM ClienteLoyaltyMov WHERE Codigo_Reserva = NEW.Codigo_Reserva) THEN

    SET v_noches = GREATEST(DATEDIFF(NEW.Fecha_Salida, NEW.Fecha_Entrada), 0);
    SET v_pts    = v_noches + ROUND(NEW.Monto_Total / 1000);

    INSERT INTO ClienteLoyaltyMov (Codigo_Cliente, Codigo_Reserva, Puntos, Monto_Base, Noches)
    VALUES (NEW.Codigo_Cliente, NEW.Codigo_Reserva, v_pts, NEW.Monto_Total, v_noches);

    UPDATE Cliente
      SET Puntos = Puntos + v_pts,
          Nivel  = loyalty_level(Puntos + v_pts),
          Descuento_Porc = CASE
            WHEN (Puntos + v_pts) >= 700 THEN 15.00
            WHEN (Puntos + v_pts) >= 300 THEN 7.50
            WHEN (Puntos + v_pts) >= 100 THEN 3.00
            ELSE 0.00
          END
      WHERE Codigo_Cliente = NEW.Codigo_Cliente;
  END IF;
END$$
DELIMITER ;

CREATE OR REPLACE VIEW v_fin_caja_resumen AS
SELECT
  s.id_session,
  s.fecha,
  s.opening_cash,
  IFNULL(r.ingresos_efectivo, 0.00) AS ingresos_efectivo,
  IFNULL(m.egresos_efectivo, 0.00)  AS egresos_efectivo,
  IFNULL(m.ajustes_mas, 0.00)       AS ajustes_mas,
  IFNULL(m.ajustes_menos, 0.00)     AS ajustes_menos,
  (s.opening_cash + IFNULL(r.ingresos_efectivo,0) - IFNULL(m.egresos_efectivo,0)
    + IFNULL(m.ajustes_mas,0) - IFNULL(m.ajustes_menos,0)) AS efectivo_esperado,
  s.closing_cash_counted AS efectivo_contado,
  (IFNULL(s.closing_cash_counted,0) -
   (s.opening_cash + IFNULL(r.ingresos_efectivo,0) - IFNULL(m.egresos_efectivo,0)
     + IFNULL(m.ajustes_mas,0) - IFNULL(m.ajustes_menos,0))) AS descuadre
FROM fin_cash_session s
LEFT JOIN (
  SELECT DATE(creado_en) AS fecha, SUM(monto) AS ingresos_efectivo
  FROM fin_receipts
  WHERE estado='Emitido' AND metodo='Efectivo'
  GROUP BY DATE(creado_en)
) r ON r.fecha = s.fecha
LEFT JOIN (
  SELECT DATE(created_at) AS fecha,
         SUM(CASE WHEN tipo='Egreso' AND metodo='Efectivo' AND estado='Aplicado' THEN monto ELSE 0 END) AS egresos_efectivo,
         SUM(CASE WHEN metodo='Efectivo' AND estado='Aplicado' AND monto >= 0 AND tipo IN ('Ingreso','Ajuste') THEN monto ELSE 0 END) AS ajustes_mas,
         SUM(CASE WHEN metodo='Efectivo' AND estado='Aplicado' AND monto < 0  AND tipo='Ajuste' THEN -monto ELSE 0 END) AS ajustes_menos
  FROM fin_cash_movement
  GROUP BY DATE(created_at)
) m ON m.fecha = s.fecha;

CREATE OR REPLACE VIEW vw_nomina_quincenal AS
SELECT
    m.Codigo_Funcionario,
    CONCAT(f.Nombre, ' ', f.Apellido) AS Colaborador,
    DATE_FORMAT(m.Fecha, '%Y-%m') AS periodo,
    CASE 
        WHEN DAY(m.Fecha) <= 15 THEN 'Q1'
        ELSE 'Q2'
    END AS quincena,
    ROUND(SUM(COALESCE(m.Horas_Regulares, 0)), 2) AS horas_aprobadas
FROM marcacion m
JOIN funcionario f 
    ON f.Codigo_Funcionario = m.Codigo_Funcionario
WHERE m.Estado = 'Aprobado'
GROUP BY
    m.Codigo_Funcionario,
    Colaborador,
    periodo,
    quincena;

SELECT *
FROM vw_nomina_quincenal
WHERE periodo = '2026-01'
  AND quincena = 'Q2'
ORDER BY Colaborador;

SELECT
    v.Codigo_Funcionario,
    v.Colaborador,
    v.periodo,
    v.quincena,
    v.horas_aprobadas,
    f.Salario_Base_Mensual,
    ROUND(f.Salario_Base_Mensual / 2, 2) AS salario_quincenal,
    ROUND((f.Salario_Base_Mensual / 2) / 92, 2) AS valor_hora,
    ROUND(((f.Salario_Base_Mensual / 2) / 92) * v.horas_aprobadas, 2) AS monto_bruto_pagar
FROM vw_nomina_quincenal v
JOIN funcionario f
  ON f.Codigo_Funcionario = v.Codigo_Funcionario
WHERE v.periodo = '2026-02'
  AND v.quincena = 'Q1';

SELECT 
    p.Id,
    f.Nombre,
    f.Apellido,
    pp.Periodo_Key,
    pp.Quincena,
    p.Bruto_Quincena,
    p.Neto_Pagar
FROM HRM_Payroll p
JOIN Funcionario f ON f.Codigo_Funcionario = p.Codigo_Funcionario
JOIN HRM_PayrollPeriod pp ON pp.Id = p.Period_Id
ORDER BY f.Nombre;
